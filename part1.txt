import { useState, useRef, useCallback, useEffect } from 'react';
import {
  Layout, Plus, Trash2, Play, ChevronLeft, ChevronRight,
  Pencil, Film, Settings, GripVertical,
  Copy, MoveUp, MoveDown, Volume2, VolumeX, Square as SquareIcon,
  Download, Layers, Upload, CheckSquare, Square, FolderArchive
} from 'lucide-react';
import JSZip from 'jszip';
import { Slide, AspectRatioKey, TransitionStyle } from './types';
import { ASPECT_RATIOS, GRADIENT_PRESETS } from './constants';
import { useAudio } from './hooks/useAudio';
import UploadModal, { type UploadedFile } from './components/UploadModal';
import SketchPad from './components/SketchPad';

function generateId() {
  return Math.random().toString(36).substring(2, 10);
}

function createSlide(sceneNo: number, shotNo: number): Slide {
  return {
    id: generateId(), sceneNo, shotNo,
    action: 'Describe the action in this shot...', dialogue: '', duration: 3, imageUrl: '', notes: '',
  };
}

function dataUrlToUint8(dataUrl: string): Uint8Array {
  const raw = atob(dataUrl.split(',')[1]);
  const arr = new Uint8Array(raw.length);
  for (let i = 0; i < raw.length; i++) arr[i] = raw.charCodeAt(i);
  return arr;
}

export default function SxaBoardTool({ onBack }: { onBack?: () => void }) {
  const [slides, setSlides] = useState<Slide[]>([]);
  const [selectedSlideId, setSelectedSlideId] = useState<string>(slides[0]?.id || '');
  const [aspectRatio, setAspectRatio] = useState<AspectRatioKey>('16:9');
  const [transitionStyle, setTransitionStyle] = useState<TransitionStyle>('fade');

  // Multi-select
  const [multiSelectedIds, setMultiSelectedIds] = useState<Set<string>>(new Set());

  // Sketch pad
  const [showSketchPad, setShowSketchPad] = useState(false);
  const singleImageInputRef = useRef<HTMLInputElement>(null);

  // Playback
  const [isPlaying, setIsPlaying] = useState(false);
  const [playbackIndex, setPlaybackIndex] = useState(0);
  const [playbackProgress, setPlaybackProgress] = useState(0);
  const [voiceoverEnabled, setVoiceoverEnabled] = useState(false);
  const playbackTimerRef = useRef<ReturnType<typeof setInterval> | null>(null);

  // UI panels
  const [showSettings, setShowSettings] = useState(false);
  const [showUploadModal, setShowUploadModal] = useState(false);

  // Zip export
  const [showZipDialog, setShowZipDialog] = useState(false);
  const [zipFileName, setZipFileName] = useState('storyboard');
  const [isExportingZip, setIsExportingZip] = useState(false);
  const lastFolderNameRef = useRef('');

  // Preview drag
  const [isPreviewDragging, setIsPreviewDragging] = useState(false);

  const { triggerAudioBeep } = useAudio();

  const selectedSlide = slides.find(s => s.id === selectedSlideId);
  const selectedIndex = slides.findIndex(s => s.id === selectedSlideId);
  const multiCount = multiSelectedIds.size;

  // ─── Multi-select ──────────────────────────────────────────────────────

  const handleSlideClick = useCallback((id: string, e: React.MouseEvent) => {
    if (e.ctrlKey || e.metaKey) {
      setMultiSelectedIds(prev => {
        const next = new Set(prev);
        if (next.has(id)) next.delete(id); else next.add(id);
        return next;
      });
      setSelectedSlideId(id);
    } else if (e.shiftKey && selectedSlideId) {
      const startIdx = slides.findIndex(s => s.id === selectedSlideId);
      const endIdx = slides.findIndex(s => s.id === id);
      if (startIdx >= 0 && endIdx >= 0) {
        const lo = Math.min(startIdx, endIdx);
        const hi = Math.max(startIdx, endIdx);
        const ids = new Set(multiSelectedIds);
        for (let i = lo; i <= hi; i++) ids.add(slides[i].id);
        setMultiSelectedIds(ids);
      }
    } else {
      setMultiSelectedIds(new Set());
      setSelectedSlideId(id);
    }
  }, [selectedSlideId, slides, multiSelectedIds]);

  const selectAllSlides = useCallback(() => {
    setMultiSelectedIds(new Set(slides.map(s => s.id)));
  }, [slides]);

  const deselectAllSlides = useCallback(() => {
    setMultiSelectedIds(new Set());
  }, []);

  const deleteMultiSelected = useCallback(() => {
    if (multiCount === 0) return;
    setSlides(prev => {
      const remaining = prev.filter(s => !multiSelectedIds.has(s.id));
      if (remaining.length > 0 && multiSelectedIds.has(selectedSlideId)) {
        setSelectedSlideId(remaining[0].id);
      }
      return remaining;
    });
    setMultiSelectedIds(new Set());
    triggerAudioBeep(200, 'sawtooth', 0.1);
  }, [multiSelectedIds, multiCount, selectedSlideId, triggerAudioBeep]);

  const moveMultiSelected = useCallback((direction: 'up' | 'down') => {
    if (multiCount === 0) return;
    setSlides(prev => {
      const indices = prev.map((s, i) => multiSelectedIds.has(s.id) ? i : -1).filter(i => i >= 0);
      if (indices.length === 0) return prev;
      const result = [...prev];
      if (direction === 'up') {
        if (indices[0] === 0) return prev;
        for (const idx of indices) {
          [result[idx - 1], result[idx]] = [result[idx], result[idx - 1]];
        }
      } else {
        if (indices[indices.length - 1] === prev.length - 1) return prev;
        for (let i = indices.length - 1; i >= 0; i--) {
          const idx = indices[i];
          [result[idx], result[idx + 1]] = [result[idx + 1], result[idx]];
        }
      }
      return result;
    });
    triggerAudioBeep(380, 'sine', 0.06);
  }, [multiSelectedIds, multiCount, triggerAudioBeep]);

  // ─── Slide management ──────────────────────────────────────────────────

  const addSlide = useCallback(() => {
    const lastSlide = slides[slides.length - 1];
    const ns = createSlide(lastSlide ? lastSlide.sceneNo : 1, lastSlide ? lastSlide.shotNo + 1 : 1);
    setSlides(prev => [...prev, ns]);
    setSelectedSlideId(ns.id);
    setMultiSelectedIds(new Set());
    triggerAudioBeep(440, 'sine', 0.1);
  }, [slides, triggerAudioBeep]);

  const deleteSlide = useCallback((id: string) => {
    setSlides(prev => {
      const f = prev.filter(s => s.id !== id);
      if (selectedSlideId === id && f.length > 0) setSelectedSlideId(f[0].id);
      return f;
    });
    triggerAudioBeep(200, 'sawtooth', 0.1);
  }, [selectedSlideId, triggerAudioBeep]);

  const duplicateSlide = useCallback((id: string) => {
    const s = slides.find(x => x.id === id);
    if (!s) return;
    const ns = { ...s, id: generateId(), shotNo: s.shotNo + 1 };
    const idx = slides.findIndex(x => x.id === id);
    setSlides(prev => [...prev.slice(0, idx + 1), ns, ...prev.slice(idx + 1)]);
    setSelectedSlideId(ns.id);
    triggerAudioBeep(520, 'sine', 0.08);
  }, [slides, triggerAudioBeep]);

  const moveSlide = useCallback((id: string, dir: 'up' | 'down') => {
    setSlides(prev => {
      const i = prev.findIndex(s => s.id === id);
      if (dir === 'up' && i > 0) { const r = [...prev]; [r[i - 1], r[i]] = [r[i], r[i - 1]]; return r; }
      if (dir === 'down' && i < prev.length - 1) { const r = [...prev]; [r[i], r[i + 1]] = [r[i + 1], r[i]]; return r; }
      return prev;
    });
    triggerAudioBeep(380, 'sine', 0.06);
  }, [triggerAudioBeep]);

  const updateSlide = useCallback((id: string, u: Partial<Slide>) => {
    setSlides(prev => prev.map(s => s.id === id ? { ...s, ...u } : s));
  }, []);

  // Single image upload
  const handleSingleImageUpload = useCallback((e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0];
    if (!file || !selectedSlideId) return;
    const reader = new FileReader();
    reader.onload = (ev) => {
      setSlides(prev => prev.map(s => s.id === selectedSlideId ? { ...s, imageUrl: ev.target?.result as string } : s));
      triggerAudioBeep(520, 'sine', 0.1);
    };
    reader.readAsDataURL(file);
    e.target.value = '';
  }, [selectedSlideId, triggerAudioBeep]);

  const handlePreviewDrop = useCallback((e: React.DragEvent) => {
    e.preventDefault();
    setIsPreviewDragging(false);
    if (!selectedSlideId) return;
    const file = e.dataTransfer.files?.[0];
    if (!file || !file.type.startsWith('image/')) return;
    const reader = new FileReader();
    reader.onload = (ev) => {
      setSlides(prev => prev.map(s => s.id === selectedSlideId ? { ...s, imageUrl: ev.target?.result as string } : s));
      triggerAudioBeep(520, 'sine', 0.1);
    };
    reader.readAsDataURL(file);
  }, [selectedSlideId, triggerAudioBeep]);

  // Playback
  const startPlayback = useCallback(() => {
    if (slides.length === 0) return;
    setPlaybackIndex(0); setPlaybackProgress(0); setIsPlaying(true);
    triggerAudioBeep(440, 'sine', 0.15);
  }, [slides.length, triggerAudioBeep]);

  useEffect(() => {
    if (!isPlaying) { if (playbackTimerRef.current) { clearInterval(playbackTimerRef.current); playbackTimerRef.current = null; } return; }
    const cs = slides[playbackIndex];
    if (!cs) { setIsPlaying(false); return; }
    const dur = cs.duration * 1000;
    const t0 = Date.now();
    if (voiceoverEnabled && (cs.action || cs.dialogue)) {
      try { window.speechSynthesis.cancel(); const u = new SpeechSynthesisUtterance(cs.dialogue || cs.action); u.rate = 0.9; window.speechSynthesis.speak(u); } catch { /* */ }
    }
    const iv = setInterval(() => {
      const el = Date.now() - t0;
      setPlaybackProgress(Math.min((el / dur) * 100, 100));
      if (el >= dur) {
        if (playbackIndex < slides.length - 1) { setPlaybackIndex(p => p + 1); triggerAudioBeep(440, 'sine', 0.05); }
        else { setIsPlaying(false); triggerAudioBeep(300, 'triangle', 0.2); if (voiceoverEnabled) window.speechSynthesis.cancel(); }
        clearInterval(iv);
      }
    }, 50);
    playbackTimerRef.current = iv;
    return () => clearInterval(iv);
  }, [isPlaying, playbackIndex, slides, triggerAudioBeep, voiceoverEnabled]);

  // Export JSON
  const exportStoryboard = useCallback(() => {
    const blob = new Blob([JSON.stringify({ aspectRatio, transitionStyle, slides }, null, 2)], { type: 'application/json' });
    const a = document.createElement('a'); a.href = URL.createObjectURL(blob); a.download = 'storyboard.json';
    document.body.appendChild(a); a.click(); document.body.removeChild(a);
  }, [aspectRatio, transitionStyle, slides]);

  // ─── ZIP export (fixed) ────────────────────────────────────────────────

  const exportZip = useCallback(async () => {
    if (slides.length === 0) return;
    setIsExportingZip(true);
    try {
      const zip = new JSZip();
      const folder = zip.folder(zipFileName || 'storyboard')!;

      for (let i = 0; i < slides.length; i++) {
        const s = slides[i];
        const num = String(i + 1).padStart(3, '0');
        const fname = `${num}_S${s.sceneNo}_Shot${s.shotNo}.png`;

        if (s.imageUrl && s.imageUrl.startsWith('data:')) {
          folder.file(fname, dataUrlToUint8(s.imageUrl), { binary: true });
        } else {
          // Generate a placeholder image
          const cw = ASPECT_RATIOS[aspectRatio].canvasWidth;
          const ch = ASPECT_RATIOS[aspectRatio].canvasHeight;
          const c = document.createElement('canvas'); c.width = cw; c.height = ch;
          const ctx = c.getContext('2d')!;
          ctx.fillStyle = '#0f172a'; ctx.fillRect(0, 0, cw, ch);
          ctx.fillStyle = '#475569'; ctx.font = 'bold 14px sans-serif'; ctx.textAlign = 'center'; ctx.textBaseline = 'middle';
          ctx.fillText(s.action || `Frame ${i + 1}`, cw / 2, ch / 2);
          folder.file(fname, dataUrlToUint8(c.toDataURL('image/png')), { binary: true });
        }
      }

      // Metadata
      folder.file('storyboard.json', JSON.stringify({
        aspectRatio, transitionStyle,
        slides: slides.map((s, i) => ({
          index: i, sceneNo: s.sceneNo, shotNo: s.shotNo,
          action: s.action, dialogue: s.dialogue, duration: s.duration, notes: s.notes,
        })),
      }, null, 2));

      const blob = await zip.generateAsync({ type: 'blob', compression: 'DEFLATE' });
      const url = URL.createObjectURL(blob);
      const a = document.createElement('a');
      a.href = url;
      a.download = `${zipFileName || 'storyboard'}.zip`;
      document.body.appendChild(a);
      a.click();
      // Cleanup after short delay
      setTimeout(() => {
        document.body.removeChild(a);
        URL.revokeObjectURL(url);
      }, 200);
      setShowZipDialog(false);
      triggerAudioBeep(600, 'sine', 0.15);
    } catch (err) {
      console.error('ZIP export failed:', err);
      alert('ZIP export failed. Check console for details.');
    } finally {
      setIsExportingZip(false);
    }
  }, [slides, zipFileName, aspectRatio, transitionStyle, triggerAudioBeep]);

  // ─── Import files ──────────────────────────────────────────────────────

  const handleImportFiles = useCallback((files: UploadedFile[], folderName?: string) => {
    const lastSlide = slides[slides.length - 1];
    let cScene = lastSlide ? lastSlide.sceneNo : 1;
    let cShot = lastSlide ? lastSlide.shotNo : 0;
    const newSlides: Slide[] = files.map((f) => {
      cShot++;
      const name = f.name.replace(/\.[^.]+$/, '').replace(/[_-]/g, ' ');
      return { id: generateId(), sceneNo: cScene, shotNo: cShot, action: name || 'Imported frame', dialogue: '', duration: 3, imageUrl: f.dataUrl, notes: f.relativePath };
    });
    setSlides(prev => [...prev, ...newSlides]);
    if (newSlides.length > 0) setSelectedSlideId(newSlides[0].id);
    setShowUploadModal(false);
    setMultiSelectedIds(new Set());
    // Track folder name for ZIP export
    if (folderName) {
      lastFolderNameRef.current = folderName;
      setZipFileName(folderName);
    }
    triggerAudioBeep(520, 'sine', 0.15);
  }, [slides, triggerAudioBeep]);

  const totalDuration = slides.reduce((sum, s) => sum + s.duration, 0);

  return (
    <div className="flex flex-col h-screen bg-slate-950 text-slate-200 overflow-hidden">
      {/* ─── Header ─── */}
      <header className="flex items-center justify-between px-5 py-3 bg-slate-900 border-b border-slate-800 shrink-0">
        <div className="flex items-center gap-3">
          {onBack && (
            <button onClick={onBack} className="p-2 bg-slate-800 hover:bg-slate-700 text-slate-400 rounded-lg transition-colors">
                <ChevronLeft className="w-4 h-4" />
            </button>
          )}
          <div className="w-8 h-8 bg-sky-600 rounded-lg flex items-center justify-center"><Film className="w-4 h-4 text-white" /></div>
          <div>
            <h1 className="text-sm font-extrabold text-white tracking-tight">SeqFrame Editor</h1>
            <p className="text-[10px] text-slate-500 font-medium">{slides.length} frames • {totalDuration}s total</p>
          </div>
        </div>
        <div className="flex items-center gap-2">
          <button onClick={() => setShowSettings(!showSettings)} className="p-2 bg-slate-800 hover:bg-slate-700 text-slate-400 rounded-lg transition-colors" title="Settings"><Settings className="w-4 h-4" /></button>
          <button onClick={() => { setShowUploadModal(true); triggerAudioBeep(380, 'sine', 0.06); }} className="flex items-center gap-1.5 px-3 py-2 bg-violet-600 hover:bg-violet-500 text-white rounded-lg text-xs font-bold transition-colors">
            <Upload className="w-3.5 h-3.5" /> Import
          </button>
          <button onClick={exportStoryboard} className="p-2 bg-slate-800 hover:bg-slate-700 text-slate-400 rounded-lg transition-colors" title="Export JSON"><Download className="w-4 h-4" /></button>
          <button onClick={() => { setShowZipDialog(true); triggerAudioBeep(380, 'sine', 0.06); }} disabled={slides.length === 0} className="flex items-center gap-1.5 px-3 py-2 bg-amber-600 hover:bg-amber-500 text-white rounded-lg text-xs font-bold transition-colors disabled:opacity-40">
            <FolderArchive className="w-3.5 h-3.5" /> Save ZIP
          </button>
          <button onClick={startPlayback} disabled={slides.length === 0} className="flex items-center gap-1.5 px-4 py-2 bg-sky-600 hover:bg-sky-500 text-white rounded-lg text-xs font-bold shadow-lg shadow-sky-500/20 disabled:opacity-40 transition-colors">
            <Play className="w-3.5 h-3.5 fill-white" /> Play
          </button>
        </div>
      </header>
